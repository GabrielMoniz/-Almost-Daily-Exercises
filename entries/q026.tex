\begin{entry}{Q026}{The hyperedge as a logical gate}

\entrymeta{

\begin{problem}
\emph{Context.} \qref{Q026} showed that Gaussian unitaries give only logical Clifford
gates on the GKP code, so universality needs a gate whose generator is at least cubic in
the quadratures. Two cubic generators suggest themselves: the three-mode
\emph{hyperedge} interaction $\hat q_1\hat q_2\hat q_3$ of the CV hypergraph states of
\qref{Q022}, and the single-mode cube $\hat q^{3}$. This entry asks which of the two
fits the code lattice.

\emph{Setting.} Use the conventions of \qref{Q025}: $\ell=\sqrt{2\pi/d}$,
$\omega=e^{2\pi i/d}$ and the ideal codewords
$\ket{\bar\jmath}=\sum_{k\in\ZZ}\ket{q=(dk+j)\ell}$, $j\in\ZZ_d$, treated formally.
The qudit controlled-controlled-$Z$ gate is
\begin{equation}
  \mathrm{CCZ}_d\ket{j_1,j_2,j_3}=\omega^{\,j_1j_2j_3}\ket{j_1,j_2,j_3},
  \qquad j_1,j_2,j_3\in\ZZ_d .
  \label{q027:ccz}
\end{equation}
For real $c$ and $\gamma$ consider
\begin{equation}
  U_c:=e^{\,ic\,\hat q_1\hat q_2\hat q_3},\qquad V_\gamma:=e^{\,i\gamma\,\hat q^{3}} .
  \label{q027:gates}
\end{equation}

A \emph{code state} is a linear combination of the codewords. Each codeword is a comb of
position spikes with equal coefficients, and different codewords have their spikes at
different positions. Both gates are functions of $\hat q$ alone, so they leave each spike
in place and only multiply its coefficient by a phase. Hence the image of a codeword is
a code state if and only if every spike of that comb receives the same phase:
\begin{equation}
  U\ket{\bar\jmath}\ \text{is a code state}\iff
  \text{the phase is the same at every spike of }\ket{\bar\jmath},
  \label{q027:reduction}
\end{equation}
and then $U\ket{\bar\jmath}$ is that phase times $\ket{\bar\jmath}$.


\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*]
  \item \emph{The hyperedge.} Find all $c>0$ for which $U_c$ maps the code space into itself, and show that the smallest such $c$ acts on the code exactly as $\mathrm{CCZ}_d$, for every $d\ge2$.
  
  \item \emph{The pure cube.}  Find every $\gamma$ for which $V_\gamma$ maps the code space into itself, and the logical gate it then implements. Take $d=2$, can any $V_\gamma$ act as the logical $T=\mathrm{diag}(1,e^{i\pi/4})$, up to a global phase?
\end{enumerate}
\end{problem}

\begin{solution}[Part (a)]
Analyzing how the gates act on each element of the comb 
\begin{equation}
\begin{split}
     U_c \ket{(d k_1+j_1)\ell , (d k_2+j_2)\ell , (d k_3+j_3)\ell }=& e^{i (d k_1+j_1) (d k_2+j_2) (d k_3+j_3) c\ell^3} \ket{(d k_1+j_1) \ell, (d k_2+j_2)\ell , (d k_3+j_3)\ell } \\
     =& \omega^{(d k_1+j_1) (d k_2+j_2) (d k_3+j_3) c\ell}  \ket{(d k_1+j_1) \ell, (d k_2+j_2)\ell , (d k_3+j_3) \ell}.
\end{split}
\end{equation}
Define $M(\mathbf k, \mathbf{j})$ such that $\prod_i (dk_i +j_i)= j_1j_2j_3 + d M(\mathbf k, \mathbf{ j})$, such that the exponent can be expressed by $(j_1j_2j_3 + dM(\mathbf k, \mathbf{ j}))c \ell$. We need any dependence on $\mathbf k$ to disappear, and this should hold for any triple $j_1, j_2, j_3$. Since $\omega$ defines a algebra modulo $d$, we can see it is sufficient to have $c \ell \in \ZZ$, which results in 
$$(j_1j_2j_3 + dM(\mathbf k, \mathbf{ j}))n - nj_1j_2j_3 = n dM(\mathbf k, \mathbf{ j})$$ being always a multiple of $d$, resulting in 
$U_c \equiv \omega^{n j_1j_2j_3}$ over the comb states. 

To see this is necessary, consider the case $\mathbf j = (1,1,1)$ and compare the comb points $\mathbf k = (0,0,0)$ such that $M(\mathbf k, \mathbf{ j})=0$, with $U_c$ producing the phase $\omega^{c\ell}$, with the comb point $k=(1,0,0)$, such that $M(\mathbf k, \mathbf{ j})=1$, with $U_c$ producing the phase $\omega^{c\ell + dc\ell}$. In order to both be the same, we need $\omega^{dc\ell}=1$, which is possible only if $c \ell \in \ZZ$.

Adding the given condition that $c>0$, the smallest possible $c$ is $1/\ell$  (since it must satisfy $c_n = n/\ell$), and in this case the phase produced in the comb states is precisely $\omega^{j_1j_2j_3}$.


\end{solution}

\begin{solution}[Part (b)]
Analyzing also the action on the comb states
\begin{equation}
\begin{split}
    V_\gamma\ket{(dk+j)\ell}  &=   e^{i  \gamma (dk +j)^3 \ell^3}  \ket{(dk+j)\ell}\\
    &=   \omega^{  (dk +j)^3 \gamma\ell}  \ket{(dk+j)\ell}
    \label{V action}
\end{split}
\end{equation}
Despite the conditions analogous to the previous $\gamma \ell$ also be sufficient, it is not necessary here for every $d$. 

To see this, take $j=1$, for $k=0$ the phase is $\omega^{\gamma \ell}$, whereas for $k=1$, the phase is $\omega^{(d+1)^3\gamma \ell}=\omega^{(d^3 +3d^2+3d+1)\gamma \ell}$, in order to them be the same we need
$\omega^{(d^3 +3d^2+3d)\gamma \ell} = e^{2\pi i (d^2+3d+3)\gamma \ell}=1$, i.e. $(d^2+3d+3)\gamma \ell \in \ZZ$. Notice how here, despite the action of the gate being also cubic, the phase difference between nearest neighbors in the lattice was linear (moving from $\mathbf k = (0,0,0)$ to $\mathbf k = (1,0,0)$ added to the exponent a linear term in $d$, $dc\ell$), whereas here, due to the cubic factor acting in only one mode, the phases between nearest neighbors $k \rightarrow k+1$ vary nonlinearly in $d$, producing the different behavior.
This nonlinearity also affects $j$ as one can see in \eqref{V action}, take $j=0$, for $k=0$ the phase is $\omega^0=1$, while for $k=1$ it is $\omega^{d^3 \gamma \ell}= e^{i2\pi d^2\gamma \ell}$, which imposes $d^2\gamma \ell \in \ZZ$. For general $d$, each $j=0,\cdots,d-1$ imposes a different constraint $N_j \gamma \ell \in \ZZ$ for some integer $N_j$. Thus, requiring to analyze differently for each $d$.

For $d=2$, the condition for $j=0$ imposes $4\gamma \ell \in \ZZ$, whereas the condition for $j=1$ imposes $13\gamma \ell \in \ZZ$ simultaneously; since $\text{mdc}(13,4)=1$, this is only possible if $\gamma\ell$ itself is an integer, which is analogous to the previous item, and sufficiency follows immediately. So we need $\gamma \equiv \gamma_n = n/\ell$ resulting in 
\begin{equation}
    V_n \ket{\bar 0} = \ket{\bar 0}, \quad V_n \ket{\bar 1}= \omega^{n}\ket{\bar 1}= (-1)^{n}\ket{\bar 1} 
 \end{equation}
 clearly not being possible to act as the $T$ gates, which requires a relative phase of $\pi/4$, but here is multiple of $\pi$.

For $d=3$, the condition for $j=0$ imposes $9\gamma\ell \in \ZZ$, for $j=1$, imposes $21 \gamma \ell \in \ZZ$, and for $j=2$, it imposes $(3^2 +3\cdot2\cdot3 + 3\cdot2^2)\gamma\ell=3\cdot13 \gamma \ell$. So, since $3$ divides all these $N_j$, we need $\gamma \ell = n/3$ in order for all these conditions to be met. To see this is sufficient, take $\gamma \equiv \gamma_n = n/3\ell$ and substitute on \eqref{V action} 
\begin{equation}
\begin{split}
    V_\gamma\ket{(dk+j)\ell} &=   \omega^{  (3k +j)^3 n/3}  \ket{(dk+j)\ell}\\
    &= \omega^{(27k^3+ 27j+ 9j^3 + j^3)n/3}  \ket{(dk+j)\ell}\\
    &= \omega^{ j^3n/3}  \ket{(dk+j)\ell}
\end{split}
\end{equation}
acting as a constant phase for a fixed $j$.


\end{solution}
