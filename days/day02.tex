\daytitle{2}{Squaring is not operator monotone}

\begin{daymeta}
  \item[Topic] Operator inequalities; matrix analysis; positivity vs.\ trace positivity.
  \item[Estimated time] $\sim 12$ minutes (solving and write-up).
  \item[Origin] Adapted. Classical material; the two-part contrast and the phrasing
    are mine.
  \item[Source] The underlying fact is the failure of operator monotonicity of
    $t\mapsto t^{2}$, the sharp complement to the L\"owner--Heinz inequality
    ($A\ge B\ge 0$ implies $A^{\alpha}\ge B^{\alpha}$ for every $\alpha\in[0,1]$, and
    $t^{\alpha}$ fails to be operator monotone for every $\alpha>1$). Standard
    treatment: R.~Bhatia, \emph{Matrix Analysis}, Springer GTM 169 (1997), Ch.~V,
    ``Operator Monotone and Operator Convex Functions''.
  \item[Prerequisites] None beyond $M\ge 0$ meaning positive semidefinite.
\end{daymeta}

\begin{problem}
Let $A,B$ be Hermitian operators on $\CC^{d}$ with $0\le A\le B$, i.e.\ $A\ge 0$ and
$B-A\ge 0$.
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*]
  \item Prove that $\Tr(A^{2})\le\Tr(B^{2})$.
  \item Prove or disprove: $A^{2}\le B^{2}$.
\end{enumerate}
If you disprove it, exhibit an explicit counterexample in the smallest dimension where
one exists, and verify all three required positivity conditions. Then state a nontrivial
hypothesis on $A$ and $B$ under which the implication \emph{does} hold, and say in one
line why your counterexample must violate it.
\end{problem}
