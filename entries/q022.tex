\begin{problem}
\emph{Context.} Continuous-variable graph states are built by applying two-mode
interactions $e^{i\hat q_j\hat q_k}$ to momentum eigenstates. They are Gaussian, so they
cannot by themselves give a computational advantage. Hypergraph states add interactions
among three or more modes, such as $e^{i\hat q_1\hat q_2\hat q_3}$, and a lattice of such
hyperedges is a resource for universal computation using only homodyne measurements.
These states are characterised by their \emph{nullifiers}, operators that annihilate the
state, and experiments certify them through the variances of the nullifiers, which are
second moments. Neither the definition nor the certification refers to the Wigner
function. This exercise computes it for a single hyperedge and asks what it reveals.

\emph{Setting.} Take $k$ bosonic modes with $[\hat q_j,\hat p_l]=i\delta_{jl}$,
$\hbar=1$, and write $q=(q_1,\dots,q_k)$, $p=(p_1,\dots,p_k)$ for points of phase space
$\RR^{2k}$. The Wigner function of \qref{Q022} extends to $k$ modes as
\begin{equation}
  W_\rho(q,p):=\frac{1}{\pi^{k}}\int_{\RR^{k}}d^{k}y\;e^{2ip\cdot y}\,
  \bra{q-y}\rho\ket{q+y},
  \label{q022:wigner}
\end{equation}
where $\ket{q}=\ket{q_1}\otimes\cdots\otimes\ket{q_k}$ are joint position
eigenstates, and Royer's identity extends as
\[
  W_\rho(q,p)=\frac{1}{\pi^{k}}\Tr\bigl[\rho\,D(q,p)\,\Pi\,D(q,p)^{\dagger}\bigr],
  \qquad
  D(q,p):=e^{\,i\sum_j(p_j\hat q_j-q_j\hat p_j)}=\bigotimes_{j=1}^{k}D_j(q_j,p_j),
\]
\[
  \Pi:=(-1)^{\sum_j a_j^{\dagger}a_j}=\bigotimes_{j=1}^{k}\Pi_j,
  \qquad
  \Pi\ket{q}=\ket{-q}.
\]
Both $D$ and $\Pi$ factorise over the modes, and both sides of the identity are linear in
$\rho$, so it follows from the single-mode case applied to product operators
$\ket{a_1}\!\bra{b_1}\otimes\cdots\otimes\ket{a_k}\!\bra{b_k}$, even for entangled $\rho$.
For a pure state, $\bra{q-y}\rho\ket{q+y}=\psi(q-y)\,\psi^{*}(q+y)$.

For a real polynomial $f$ on $\RR^{k}$ consider the (unnormalisable) \emph{phase state}
\begin{equation}
  \ket{\psi_f}:=e^{if(\hat q)}\ket{\Omega}^{\otimes k},
  \qquad
  \ket{\Omega}:=\int_{\RR}dq\,\ket{q}=\sqrt{2\pi}\,\ket{p=0},
  \qquad
  \braket{q}{\psi_f}=e^{if(q)} .
  \label{q022:phase-state}
\end{equation}
Since the state is not normalisable, $W$ is a distribution. You may manipulate the
integrals formally, exchanging their order and using
$\int_{\RR}dy\,e^{iky}=2\pi\,\delta(k)$. Since
$\hat p_i\ket{\psi_f}=\partial_i f(\hat q)\ket{\psi_f}$, the state is annihilated
by the \emph{nullifiers}
\begin{equation}
  \hat n_i:=\hat p_i-\partial_i f(\hat q),
  \label{q022:nullifier}
\end{equation}
and we write $n_i(q,p):=p_i-\partial_i f(q)$ for the corresponding functions on phase
space (their Weyl symbols). For $f=q_1q_2$ (one graph-state edge) and $f=q_1q_2q_3$ (one
hyperedge) this gives $n_i=p_i-\prod_{j\neq i}q_j$.

You may use the following identity, which follows from the integral
representations of the Bessel functions $Y_0$ and $K_0$ (NIST DLMF, \S10.9 and
\S10.32) after the substitution $t=e^{u}$. For real $a,b$ with $ab\neq0$,
\begin{equation}
  \int_{0}^{\infty}\frac{dt}{t}\,\cos\!\Bigl(at+\frac{b}{t}\Bigr)
  =\begin{cases}
     -\pi\,Y_0\bigl(2\sqrt{ab}\bigr), & ab>0,\\[2pt]
     2\,K_0\bigl(2\sqrt{-ab}\bigr), & ab<0 .
   \end{cases}
  \label{q022:bessel}
\end{equation}
Here $K_0(x)>0$ for all $x>0$, decays like $e^{-x}$, and diverges like $-\ln x$ as
$x\to0^{+}$. $Y_0(x)$ diverges like $\tfrac{2}{\pi}\ln x$ as $x\to0^{+}$ and oscillates
with decaying amplitude; its first zeros are $x\approx0.894,\ 3.958,\ 7.086$, and it is
positive between the first two.

\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*]
  \item \emph{One edge.} Show that for a pure phase state
        \begin{equation}
          W(q,p)=\frac{1}{\pi^{k}}\int d^{k}y\;
          \exp\!\Bigl(i\bigl[2p\cdot y-f(q+y)+f(q-y)\bigr]\Bigr).
          \label{q022:phase-wigner}
        \end{equation}
        Evaluate $W$ for $f=q_1q_2$ and describe it in terms of the nullifiers.

  \item \emph{Phase states in general.} Using a Taylor expansion of the one-variable
        function $t\mapsto f(q+ty)$, derive an exact expression for $f(q+y)-f(q-y)$,
        valid for a polynomial $f$ of any degree, and insert it into
        \eqref{q022:phase-wigner}. Deduce that
        $W(q,p)=\prod_i\delta\bigl(n_i(q,p)\bigr)$ holds for every quadratic $f$.

  \item \emph{The hyperedge.} Let $f=q_1q_2q_3$. Using (b), reduce $W$ to a
        one-dimensional integral, and using the identity \eqref{q022:bessel}, write
        $W$ in closed form. Show that $W$ depends on the six phase-space
        coordinates only through a single function of $n_1,n_2,n_3$.

  \item \emph{Where the non-Gaussianity lives.} Identify, in terms of the nullifiers,
        the set on which $W$ is singular and the region where it is negative. In (a),
        $W$ vanishes outside $\{n_1=n_2=0\}$. Show that
        $\int d^{k}p\,W\,n_i^{2}=0$ at every $q$ for every phase state, and explain
        why, for the hyperedge, this is compatible with the closed form of (c) only
        because $W$ takes negative values.

  \item \emph{Measuring one mode (Lemma~1 in phase space).} A homodyne measurement of
        $\hat q_3$ with outcome $m$ leaves modes $1$ and $2$ in the unnormalised state
        with wavefunction $\psi_m(q_1,q_2):=\psi(q_1,q_2,m)$. Show from the definition
        \eqref{q022:wigner} that its Wigner function is
        \begin{equation}
          W_m(q_1,q_2;p_1,p_2)=\int dp_3\,W(q_1,q_2,m;p_1,p_2,p_3),
          \label{q022:conditional}
        \end{equation}
        that is, the measured coordinate is set to the outcome and its conjugate
        momentum is integrated out. For the hyperedge, compute $W_m$ from the
        three-dimensional integral form of (c), without using the closed form. Which
        state of modes $1$ and $2$ is this, and is it Gaussian? The three-mode $W$ is
        negative on part of phase space and $W_m$ is not: which term of the exponent
        carries the negativity, and why does the integration over $p_3$ remove it?
\end{enumerate}
\end{problem}

\begin{solution}[Part (a)]
The integral form derives directly from Eq.~\eqref{q022:wigner} and by $\braket{q}{\psi_f}=e^{if(q)}$, Eq.~\eqref{q022:phase-state}. For the graph state

\begin{equation}
\begin{split}
    W(q,p)&= \frac{1}{\pi^2} \int d^2y \exp\left(i[2p_1y_1 +2p_2y_2- (q_1+y_1)(q_2+y_2)+(q_1-y_1)(q_2-y_2)]\right)\\
    &= \frac{1}{\pi^2} \int d^2y \exp\left(2i[p_1y_1 +p_2y_2- q_1y_2- q_2y_1]\right)\\
    &= \frac{1}{\pi^2} \int d^2y \exp\left(2i[y_1\ n_1 +y_2\ n_2]\right)\\
    &= \left(\frac{1}{\pi} \int dy \exp(2iyn_1)\right) \left(\frac{1}{\pi} \int dy \exp(2iyn_2)\right)\\
    &= \left(\frac{1}{2\pi} \int dy \exp(iyn_1)\right) \left(\frac{1}{2\pi} \int dy \exp(iyn_2)\right)\\
    &= \delta(n_1)\delta(n_2)
\end{split}
\label{q022:graph-wigner}
\end{equation}
The crucial step is that $f(q+y)-f(q-y)$ kills the quadratic terms on either $q$ or $y$ for $f=q_1q_2$, leaving only the cross terms. This allows this simple form in terms of the nullifier with factorized integrals.
\end{solution}

\begin{solution}[Part (b)]
Fix $q,y \in \RR^k$ and define $g(t)=f(q+ty)$. If $f$ is of degree $r$, then $g$ itself is a one-variable polynomial of degree $r$. Moreover, one can Taylor expand in the real variable $t$ to see
\begin{equation}
    g(t)= \sum_{m=0}^r \frac{g^{(m)}(0)}{m!} t^m= \sum_{m} \frac{1}{m!} \left(y\cdot\nabla \right)^mf(q)  t^m
\end{equation}
for
\[
  \nabla = \left(\frac{\partial}{\partial q_1},\cdots,\frac{\partial}{\partial q_k}\right)\equiv\left(\partial_1,\cdots,\partial_k\right).
\]

Then, noticing that $f(q+y)-f(q-y)=g(1)-g(-1)$, one can see that the even terms in $y$ cancel. Only the odd terms survive, where taking $u=\lfloor  \frac{r-1}{2}\rfloor$,
\begin{equation}
    f(q+y)-f(q-y) = 2\sum_{l=0}^{u} \frac{1}{(2l+1)!} \left(y\cdot\nabla \right)^{2l+1}f(q)
    \label{q022:odd-expansion}
\end{equation}
(but we can drop the $u$ and consider the summation to infinity, since for $l>u$, we have $\left(y\cdot\nabla \right)^{2l+1}f(q)=0$ for any $y$, because $f$ is a polynomial of degree $r$).

For a quadratic polynomial $f$, this is simply $2(y\cdot\nabla)f(q)$, substituting into the integral of the Wigner function \eqref{q022:phase-wigner} we get
\begin{equation}
\begin{split}
     W(q,p)&=\frac{1}{\pi^{k}}\int d^{k}y\;
          \exp\!\Bigl(2i\bigl[p\cdot y-(y\cdot\nabla)f(q)\bigr]\Bigr)\\
          &=\frac{1}{\pi^{k}}\int d^{k}y\;
          \exp\!\Bigl(2i y\cdot \bigl[p-\nabla f(q)\bigr]\Bigr)\\
          &=\frac{1}{\pi^{k}}\int d^{k}y\;
          \exp\!\Bigl(2i \sum_j y_j \bigl(p_j-\partial_j f(q)\bigr)\Bigr)\\
          &= \prod_{j=1}^{k} \frac{1}{2\pi} \int dy \exp(iy n_j(q,p))\\
          &= \prod_{j=1}^{k} \delta(n_j(q,p))
\end{split}
\label{q022:quadratic-wigner}
\end{equation}

So we can see that for any quadratic $f$ the Wigner function is non-negative.
\end{solution}

\begin{solution}[Part (c)]
For a degree $3$ polynomial, given the difference expression \eqref{q022:odd-expansion}, we must take into account the previous first order derivative, but also
\begin{equation*}
    2 \frac{1}{3!} \left(y\cdot\nabla \right)^{3}f(q)
\end{equation*}
which is just added to the exponent
\begin{equation}
     W(q,p)=\frac{1}{\pi^{k}}\int d^{k}y\;
          \exp\!\Bigl(2i\bigl[p\cdot y-(y\cdot\nabla)f(q)- \frac{1}{3!} \left(y\cdot\nabla \right)^{3}f(q) \bigr]\Bigr).
     \label{q022:cubic-wigner}
\end{equation}

Calculating the third-order derivative for the given function that represents only one hyperedge $f=q_1q_2q_3$, we can see that the only combination of partial derivatives that survives are
$\partial_1\partial_2\partial_3$ and its $3!$ permutations ($f$ is smooth).
Thus, the Wigner function reduces to
\begin{equation}
\begin{split}
   W(q,p)&=\frac{1}{\pi^{3}}\int d^{3}y\;
          \exp\!\Bigl(2i\bigl[ y_1n_1 + y_2n_2 + y_3n_3-  y_1y_2y_3 \bigr]\Bigr)\\
          &= \frac{1}{\pi^{3}}\int dy_3dy_2\;\exp\left(2i(y_2n_2+y_3n_3)\right)
          \int dy_1\exp\!\Bigl(2i y_1\bigl[  n_1-  y_2y_3 \bigr]\Bigr)\\
          &= \frac{1}{\pi^{2}}\int dy_3dy_2\;\exp\left(2i(y_2n_2+y_3n_3)\right)
          \left(\frac{1}{2\pi}\int dx\exp\!\Bigl(i x\bigl[  n_1-  y_2y_3 \bigr]\Bigr)\right)\\
          &= \frac{1}{\pi^{2}}\int dy_3dy_2\;\exp\left(2i(y_2n_2+y_3n_3)\right)
          \delta(n_1-y_2y_3)\\
          &= \frac{1}{\pi^{2}}\int dy_3dy_2\;\exp\left(2i(y_2n_2+y_3n_3)\right)
          \frac{1}{|y_3|}\delta\left(\frac{n_1}{y_3}-y_2\right)\\
          &= \frac{1}{\pi^{2}}\int dx\;\exp\left(2i\left(\frac{n_1n_2}{x}+xn_3\right)\right)
          \frac{1}{|x|}
\end{split}
\label{q022:reduction}
\end{equation}

We can separate the integral in the regions $x<0$ and $x>0$, and make $x\rightarrow -x$ over the integral in negative domain
\begin{equation}
\begin{split}
   W(q,p)&= \frac{1}{\pi^{2}}\int^{\infty}_0 dx\;\exp\left(2i\left(\frac{n_1n_2}{x}+xn_3\right)\right)
          \frac{1}{x} - \frac{1}{\pi^{2}}\int_{-\infty}^0 dx\;\exp\left(2i\left(\frac{n_1n_2}{x}+xn_3\right)\right)
          \frac{1}{x}\\
          &= \frac{1}{\pi^{2}}\int^{\infty}_0 dx\;\exp\left(2i\left(\frac{n_1n_2}{x}+xn_3\right)\right)
          \frac{1}{x} -\frac{1}{\pi^{2}}\int_{\infty}^0 dx\;\exp\left(-2i\left(\frac{n_1n_2}{x}+xn_3\right)\right)
          \frac{1}{x}\\
          &= \frac{1}{\pi^{2}}\int_{0}^{\infty} dx\;\left[\exp\left(2i\left(\frac{n_1n_2}{x}+xn_3\right)\right) + \exp\left(-2i\left(\frac{n_1n_2}{x}+xn_3\right)\right) \right] \frac{1}{x}\\
          &= \frac{2}{\pi^{2}}\int_{0}^{\infty}
          dx\;\cos\left(2\frac{n_1n_2}{x}+2xn_3 \right)  \frac{1}{x}
\end{split}
\label{q022:cosine-form}
\end{equation}

Thus, from the identity of the Bessel Functions \eqref{q022:bessel}, the Wigner Function can be written as
\begin{equation}
   W(q,p) =\begin{cases}
     -\frac{2}{\pi}\,Y_0\bigl(4\sqrt{n_1n_2n_3}\bigr), & n_1n_2n_3>0,\\[2pt]
     \frac{4}{\pi^2}\,K_0\bigl(4\sqrt{-n_1n_2n_3}\bigr), & n_1n_2n_3<0 .
   \end{cases}
   \label{q022:closed-form}
\end{equation}
So the formula, in a surprisingly interesting way, reduces to a function of the product of the (Weyl symbols of) the nullifiers $n_1n_2n_3$.

\end{solution}

\begin{solution}[Part (d)]

Since $K_0(x)$ is always positive, the Wigner function is positive whenever $n_1n_2n_3$ is negative. The Wigner function can only be negative in the region where $n_1n_2n_3>0$, and in that region it has the opposite sign of $Y_0$, then it is negative if and only if $Y_0$ is positive.

Labeling $x_m$ the $m$-th zero of $Y_0$, it is given that $Y_0(x)>0,  x_1<x<x_2$. The oscillatory behavior around the $x$ axis implies $Y_0(x)$ is positive in the open intervals $(x_{1+2l} , x_{2+2l})$, i.e., the Wigner function is negative whenever $x_{1+2l}<4\sqrt{n_1n_2n_3}<x_{2+2l}$ for some $l\geq 0$. And it is clearly singular at $n_1n_2n_3=0$, since the integral diverges.

Now to show that the second moment of the nullifier always vanishes for any phase state we must recall \eqref{q022:phase-wigner} is such that the integrand can be split into the product of one function that depends purely on the phase momentum $p$ and other on $q$.
\[
  e^{2ip\cdot y} e^{i(f(q-y)-f(q+y))}.
\]

From the expression \eqref{q022:odd-expansion} for $f(q+y)-f(q-y)$, we can separate the lowest order term of the Taylor expansion and express the remaining terms as an arbitrary function that depends only on the $q,y$, not on $p$, $R(q,y)$
\[
  f(q+y)-f(q-y)= 2 (y\cdot\nabla)f(q) + R(q,y),
\]
such that the integrand becomes
\begin{equation}
\begin{split}
    &e^{2ip\cdot y} e^{-i[2 (y\cdot\nabla)f(q) + R(q,y)]}= e^{2iy\cdot(p-\nabla f(q))} e^{-iR(q,y)}\ ;\\
    &R(q,y)=  2\sum_{l=1}^{u} \frac{1}{(2l+1)!} \left(y\cdot\nabla \right)^{2l+1}f(q)
\end{split}
    \label{q022:remainder}
\end{equation}

Then the relation with the nullifier becomes explicit, since $y\cdot(p-\nabla f(q))= \sum_j y_j n_j$, such that
\[
  \frac{\partial^2}{\partial y_j^2} e^{2iy\cdot(p-\nabla f(q))}= -4n_j^2 e^{2iy\cdot(p-\nabla f(q))}.
\]

Then, to finally start the demonstration, another observation required. For $q=q_0$ held constant, integrating over $dp_i$ is equivalent to integrating over $dn_i$ (we are integrating from $-\infty$ to $\infty$ and subtracting by $\partial_jf(q_0)$ is subtracting by a constant).

So starting from the desired expression and using our facts
\begin{equation}
\begin{split}
    \int d^{k}p\,W(q_0,p)\,n_j^{2}(q_0,p)&= \int d^{k}n\,W(q_0,p)\,n_j^{2}(q_0,p) \\
    &= \frac{1}{\pi^{k}}\int d^{k}n\, \int d^{k}y\;
          \exp\!\Bigl(i\bigl[2p\cdot y-f(q_0+y)+f(q_0-y)\bigr]\Bigr)\,n_j^{2}\\
    &= \frac{1}{\pi^{k}} \int d^{k}y\, \int d^{k}n\
          e^{-iR(q_0,y)}\, e^{2iy\cdot(p-\nabla f(q_0))} n_j^{2} \\
    &= -\frac{1}{4\pi^{k}} \int d^{k}y\, e^{-iR(q_0,y)} \int d^{k}n\
          \, \frac{\partial^2}{\partial y_j^2}e^{2iy\cdot(p-\nabla f(q_0))} \\
    &=- \frac{1}{4\pi^{k}} \int d^{k}y\, e^{-iR(q_0,y)}\frac{\partial^2}{\partial y_j^2} \int d^{k}n \, e^{2iy\cdot n}\\
    &= - \frac{1}{4} \int d^{k}y\, e^{-iR(q_0,y)}\frac{\partial^2}{\partial y_j^2} \frac{1}{2^k \pi^k} \int d^{k}n \, e^{iy\cdot n}\\
    &=- \frac{1}{4} \int d^{k}y\, e^{-iR(q_0,y)}\frac{\partial^2}{\partial y_j^2} \delta^{k}(y)
\end{split}
\label{q022:moment-delta}
\end{equation}

Now to deal with this final integral, one must notice that the delta function is $k$-dimensional, forcing all others $y_i=0$, for $i\neq j$, in $R(q_0,y)$. From its expression this reduces the remainder \eqref{q022:remainder} to
\begin{equation}
     R(q_0, y)|_{y_i=0,i\neq j} =  2\sum_{l=1}^{u} \frac{1}{(2l+1)!} y_j^{2l+1} \cdot \partial_j^{2l+1}f(q_0).
\end{equation}
To avoid this overcrowded notation, define
\begin{equation}
    \beta_j(x) = 2\sum_{l=1}^{u} \frac{1}{(2l+1)!} x^{2l+1} \cdot \partial_j^{2l+1}f(q_0).
    \label{q022:beta}
\end{equation}
(where $u$ again is related to the degree of the polynomial $f$, $r$, by $u =\lfloor \frac{r-1}{2}\rfloor$) such that $\beta_j(y_j)=R(q_0, y)|_{y_i=0,i\neq j}$, we can rewrite our integral as
\begin{equation}
    \int d^{k}p\,W(q_0,p)\,n_j^{2}= - \frac{1}{4} \int dx\, e^{-i\beta_j(x)}\frac{\partial^2}{\partial x^2} \delta(x)
\end{equation}

To deal with this (rather messy) ``derivative of the delta function'', which can still be defined as a distribution through its operation on functions, by applying integration by parts.
The first derivative acts as, noticing the boundary terms vanish
\begin{equation}
    \int dx\ g(x)\delta'(x) = - \int dx\ g'(x)\delta(x)=-g'(0)
\end{equation}
and similarly
\begin{equation}
    \int dx\ g(x)\delta''(x) = - \int dx\ g'(x)\delta'(x)=g''(0)
\end{equation}

Then our beautiful integral becomes
\begin{equation}
\begin{split}
    \int d^{k}p\,W(q_0,p)\,n_j^{2}&= - \frac{1}{4} \left(\frac{d^2}{dx^2} e^{-i\beta_j(x)}\right)_{x=0} \\
    &= \frac{1}{4} e^{-i\beta_j(0)} \left[i\frac{d^2}{dx^2} \beta_j + \left(\frac{d}{dx} \beta_j\right)^2 \right]_{x=0}
\end{split}
\label{q022:second-moment}
\end{equation}
and from \eqref{q022:beta} one can see this is always zero, since either $\beta$ itself is already zero, as is the case for quadratic polynomials, where we don't have a remainder in \eqref{q022:remainder}, or the lowest degree on $x$ is at least three, hence both the first and the second derivative have no constant terms that survive when $x=0$ (the lowest degree is at least two for the first derivative and at least one for the second derivative).

This null second moment of the nullifier forces a negativity of the Wigner function every time the set of values the Wigner function is non-zero for $n_j \neq 0$ has non-zero measure.

For the quadratic case, we have a delta function that vanishes whenever $n_j\neq 0$, while for the hypergraph state the Wigner function is non-zero almost everywhere, so to counter-balance the positive values when integrating $W(q_0,p)n_j(q_0,p)^2$, we must have negative ones also.

\end{solution}

\begin{solution}[Part (e)]
Start from the definition applying directly to the wave function $\psi_m$
\begin{equation}
\begin{split}
    W_m(q_1,q_2,p_1,p_2) &= \frac{1}{\pi^{2}}\int_{\RR^{2}}d^{2}y\;e^{2i(p_1y_1+p_2y_2)}\,
  \psi_m(q_1-y_1,q_2-y_2)\psi^*_m(q_1+y_1,q_2+y_2) \\
  &= \frac{1}{\pi^{2}}\int_{\RR^{2}}d^{2}y\;e^{2i(p_1y_1+p_2y_2)}\,
  \psi(q_1-y_1,q_2-y_2,m)\psi^*(q_1+y_1,q_2+y_2,m) \\
  &= \frac{1}{\pi^{2}}\int_{\RR^{2}}d^{2}y\ dx\;e^{2i(p_1y_1+p_2y_2)}\,
  \psi(q_1-y_1,q_2-y_2,m-x)\psi^*(q_1+y_1,q_2+y_2,m+x)\delta(x)
\end{split} 
\end{equation}
where the last equation is valid since the delta forces $x=0$ (integral version of adding 0).

On the other side, $\int dp_3 e^{2ip_3y_3}=\pi\delta(y_3)$, thus
\hspace{-2cm}
\begin{equation}
\begin{split}
    &W_m(q_1,q_2,p_1,p_2) \\
    &= \frac{1}{\pi^{2}}\int_{\RR^{3}}d^{3}y\;e^{2i(p_1y_1+p_2y_2)}\,
  \psi(q_1-y_1,q_2-y_2,m-y_3)\psi^*(q_1+y_1,q_2+y_2,m+y_3)\delta(y_3)\\
  &= \frac{1}{\pi^{3}}\int_{\RR^{3}}d^{3}y\;e^{2i(p_1y_1+p_2y_2)}\,
  \psi(q_1-y_1,q_2-y_2,m-y_3)\psi^*(q_1+y_1,q_2+y_2,m+y_3)\int dp_3 e^{2ip_3y_3}\\
  &= \int dp_3 \frac{1}{\pi^{3}}\int_{\RR^{3}}d^{3}y\;e^{2i(p_1y_1+p_2y_2+p_3y_3)}\,
  \psi(q_1-y_1,q_2-y_2,m-y_3)\psi^*(q_1+y_1,q_2+y_2,m+y_3)\\
  &= \int dp_3\,W(q_1,q_2,m;p_1,p_2,p_3) 
\end{split} 
\end{equation}

So applying this to the Wigner function for the hypergraph deduced in item c), with the cubic term on $y$, and again using that for fixed $q=(q_1,q_2,m)$, $dn_i=dp_i$, (more specifically, using that $dn_3=dp_3$) and denoting $q'=(q_1,q_2)$ and $p'=(p_1,p_2)$ for the corresponding two-mode phase space variables

\begin{equation}
    \begin{split}
    W_m(q',p')&=\frac{1}{\pi^{3}}\int d^{3}y\;dn_3
          \exp\!\Bigl(2i\bigl[ y_1n_1 + y_2n_2 + y_3n_3-  y_1y_2y_3 \bigr]\Bigr)\\
        &=\frac{1}{\pi^{3}}\int d^{3}y\;dn_3 e^{2iy_3n_3}
          \exp\!\Bigl(2i\bigl[ y_1n_1 + y_2n_2 -  y_1y_2y_3 \bigr]\Bigr)\\
    &= \frac{1}{\pi^{2}}\int d^{3}y\;\delta(y_3)
          \exp\!\Bigl(2i\bigl[ y_1n_1 + y_2n_2 -  y_1y_2y_3 \bigr]\Bigr)\\
          &=  \frac{1}{\pi^{2}}\int d^{2}y\;
          \exp\!\Bigl(2i\bigl[ y_1n_1 + y_2n_2  \bigr]\Bigr)\\
          &= \delta(n_1)\delta(n_2)
    \end{split}
\end{equation}
Note that the information about $m$ is encoded both on $n_1=p_1-mq_2$ and $n_2=p_2 - q_1m$. So the measurement acts by destroying the cubic interaction (with the delta function), producing a weighted graph state, with weight $m$. 

The weighted graph is a notion you can make general by associating with each edge $e$ a weight $m_e$. For this weighted CV graph state, that would correspond to the phase function $f=mq_1q_2$, which is the phase of item b) multiplied by a constant $m$. Notice that, from the general for the nullifier formula $n_1 = p_1 - \partial_1f(q)=p_1-mq_2$.



In item b), we already demonstrated that any quadratic phase $f(q)$ leads to a non-negative Wigner function, which is a product of delta functions. Negativity for a phase state can only exist for polynomials of degree at least three. To make this even clearer, one can write equation \eqref{q022:phase-wigner} in terms of the remainder \eqref{q022:remainder}
\begin{equation}
\begin{split}
    W(q,p)
    &= \frac{1}{\pi^{k}}\, \int d^{k}y\;
          \exp\!\Bigl(i\bigl[2p\cdot y-f(q+y)+f(q-y)\bigr]\Bigr)\\
    &= \frac{1}{\pi^{k}} \int d^{k}y\,
          e^{-iR(q,y)}\, e^{2iy\cdot(p-\nabla f(q))}
\end{split}
\label{q022:wigner-remainder}
\end{equation}
which explicitly shows that the existence of a non-null Remainder is what prevents the Wigner function from reducing to a product of delta functions.
